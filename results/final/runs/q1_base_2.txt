SELECT
    b.tconst,
    b.primaryTitle,
    b.startYear,
    r.averageRating,
    r.numVotes
FROM title_basics b
JOIN title_ratings r ON r.tconst = b.tconst
WHERE b.titleType = 'movie'
  AND b.startYear BETWEEN @decade_start AND @decade_end
  AND r.numVotes >= @min_votes
ORDER BY r.averageRating DESC, r.numVotes DESC, b.tconst ASC
LIMIT 20
